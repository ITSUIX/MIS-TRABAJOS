create table comentarios ( id bigint generated by default as identity primary key, nombre text not null, comentario text not null, creado_en timestamptz default now() ); alter table comentarios enable row level security; -- Cualquier visitante puede ENVIAR comentarios create policy "Cualquiera puede enviar comentarios" on comentarios for insert to anon with check ( length(nombre) between 1 and 100 and length(comentario) between 1 and 2000 ); -- Nadie que visite la web puede leer los comentarios -- No creamos ninguna política SELECT para anon. -- Solo usuarios autenticados pueden leerlos create policy "Solo administrador puede leer comentarios" on comentarios for select to authenticated using (true); -- Solo usuarios autenticados pueden eliminar comentarios create policy "Solo administrador puede eliminar comentarios" on comentarios for delete to authenticated using (true);
